```
class Device:
    def turn_on(self):
        return "Пристрій включений"

class Smartphone(Device):
    def turn_on(self):
        return "Смартфон включений"

class Laptop(Device):
    def turn_on(self):
        return "Ноутбук включений"

devices = [Smartphone(), Laptop()]

for d in devices:
    print(d.turn_on())
```

3. Реалізувати декілька “магічних методів” для роботи з визначеними раніше 
класами.

```
class Smartphone:
    def __init__(self, brand, model, price):
        self.brand = brand
        self.model = model
        self.price = price

    def __str__(self):
        return f"Смартфон {self.brand} {self.model}"

    def __lt__(self, other):
        return self.price < other.price

    def __add__(self, other):
        return self.price + other.price

p1 = Smartphone("Apple", "iPhone 15", 40000)
p2 = Smartphone("Samsung", "S23", 35000)

print(p1)

if p2 < p1:
    print(f"{p2.model} дешевший за {p1.model}")

total_cost = p1 + p2
print(f"Сумарна вартість: {total_cost} грн.")
```
